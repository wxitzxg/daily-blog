RSV:=(C-LLV(L,9))/(HHV(H,9)-LLV(L,9))\*100;

K:=SMA(RSV,3,1);

D:=SMA(K,3,1);

J:=3\*K-2\*D;

KDJ条件:=J>K AND K>50;

ZHANGFU:=(C/REF(C,50)-1)\*100 < 25 AND (C/REF(C,10)-1)\*100 < 15;

CXHZB:=EMA((EXPMA(C\*V,3)/EXPMA(V,3)+EXPMA(C\*V,6)/EXPMA(V,6)+EXPMA(C\*V,12)/EXPMA(V,12)+EXPMA(C\*V,24)/EXPMA(V,24))/4,13);

TUPO:=CROSS(C,1.05\*CXHZB);

TP1:=MA(C,5)>MA(C,10) AND MA(C,10)>MA(C,20) AND C>MA(C,60);

TP2:=C/REF(C,1)>1.045 AND C>O;

主升浪大阳突破:TP1 AND TP2 AND ZHANGFU AND TUPO AND KDJ条件 AND V>REF(V,1)\*1.6 AND C/REF(C,1)<1.095;
